2.4

The difference between single and double average CTR is akin to the ratio of averages and average of ratios. With the single average CTR, we essentially take a ratio of averages (the total clicks divided by total impressions), which means power users dominate the metric. On the other hand, double average CTR takes the average of ratios (average of individual clicks / impressions), treating each user as an individual unit. This is evidenced by the higher value for the single average CTR (0.05088508) compared to the double average CTR (0.04293482), demonstrating how heavy users (users with higher CTRs than the typical user) pull the average up when it is weighted by impressions. The single average is useful when wanting to match business reports (total clicks over total impressions), but the unit mismatch inflates false positives if the variance is not corrected (with approaches like the delta method or bootstrapping). The double average is useful because it requires no correction, and reflects a typical user’s experience more accurately when each user is weighted equally. Conversely, the double average means users with few impressions have noisy, extreme CTRs, but still count as much as heavy users.

# group assignment
group_assignment <- function(data) {
  group_a_ids <- sample(data$X, nrow(data) / 2, replace = FALSE)
  data <- data %>% 
    mutate(
      group = ifelse(X %in% group_a_ids, "A", "B")
    )
  return(data)
}

# p-value calculation
compute_pval <- function(data) {
  p_A <- data$p[data$group == "A"]
  p_B <- data$p[data$group == "B"]
  var_A <- data$variance[data$group == "A"]
  var_B <- data$variance[data$group == "B"]
  z_stat <- (p_A - p_B) / sqrt(var_A + var_B)
  p_val <- 2 * pnorm(abs(z_stat), lower.tail = FALSE)
  return(p_val)
}

# repeat simulation
simulate_1000 <- function(AA_test, title) {
  p_values <- replicate(1000, AA_test())
  hist(p_values, main = title, breaks = 20)
  mean(p_values < .05)
}

# AA test with variance approach 1, single average
AA_single_1 <- function() {
  df <- group_assignment(df)
  
  group_data <- df %>%
  group_by(group) %>%
  summarise(
    p = sum(clicks) / sum(impressions),
    variance = (p * (1 - p)) / sum(impressions)        
  )

  p_val <- compute_pval(group_data)
}

# AA test with variance approach 2, single average
AA_single_2 <- function() {
  df <- group_assignment(df)
  
  group_data <- df %>%
    group_by(group) %>%
    summarise(
      N = n(),
      avg_clicks = mean(clicks),
      avg_impressions = mean(impressions),
      p = avg_clicks / avg_impressions,
      var_clicks = var(clicks),
      var_impressions = var(impressions),
      cov_clicks_impressions = cov(clicks, impressions),
      variance = (1 / N) * (
        (var_clicks / (avg_impressions^2)) + 
          ((avg_clicks^2 * var_impressions) / (avg_impressions^4)) - 
          (2 * avg_clicks * cov_clicks_impressions / (avg_impressions^3))
      )
    )
    
    p_val <- compute_pval(group_data)
}

# repeat simulation

# bootstrapping
boot_var_single <- function(clicks, impressions, B = 200) {
  boot_ctrs <- replicate(B, {
    idx <- sample(length(clicks), replace = TRUE) 
    mean(clicks[idx]) / mean(impressions[idx])
  })
  var(boot_ctrs)
}

}
